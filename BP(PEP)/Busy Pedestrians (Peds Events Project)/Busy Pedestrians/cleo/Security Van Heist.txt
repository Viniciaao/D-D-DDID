                095F: get_car car_security door 5 angle_to 2@

                    1@ <= -0.7

                end

                    08A7:   car car_security componentB 4 opened_or_not_present

                    gosub @Open_Door

                end
            end
        end

:Damage_Car

    003B:   1@ == $PLAYER_ACTOR  // Что стреляет гг
    intensity_damage > 0.1

    check_damage = true
end

:Position_Shoot
0A97: 2@ = car car_security struct
0D4E: 2@ = struct 2@ offset 0x18 size 4
if 16@ == 0 // В for происходит смена
then 0AA7: call_function 0x4C5400 num_params 2 pop 2 "door_rr_dummy" 2@ 3@
else 0AA7: call_function 0x4C5400 num_params 2 pop 2 "door_lr_dummy" 2@ 3@
end
    if
        3@ <> 0
    then
        3@ += 0x10
        0D4E: 12@ = struct 3@ offset 0x30 size 4
        0D4E: 13@ = struct 3@ offset 0x34 size 4
        0D4E: 14@ = struct 3@ offset 0x38 size 4
        13@ -= 0.5
    else
        12@ = 0.0
        13@ = -3.0
        14@ = 1.0
    end
return

:Open_Door

repeat
wait 0
    if
        056E:   car car_security defined
    then
        4@ += 0.05 // Открываем двери
        08A6: set_car car_security door 4 rotation_to 4@
        08A6: set_car car_security door 5 rotation_to 4@
        if
            4@ > 0.9
        then
            // Процесс с выкдиваением кейса
            
            16@ = 0
            gosub @Position_Shoot
            13@ -= 1.5
            0407: store_coords_to 4@ 5@ 6@ from_car car_security with_offset 12@ 13@ 14@
            02CE: 6@ = ground_z_at 4@ 5@ 6@
            6@ += 0.5
            
            0A97: 13@ = car car_security struct
            0D4E: 13@ = struct 13@ offset Memory_adr size 1
            if
                13@ == 0
            then
                0AF0: 2@ = read_int_from_ini_file "cleo\Busy Pedestrians.ini" section "Security Van Heist" key "Money Parked Min"
                0AF0: 3@ = read_int_from_ini_file "cleo\Busy Pedestrians.ini" section "Security Van Heist" key "Money Parked Max"
                0209: 14@ = random_int_in_ranges 2@ 3@
            else
                    if
                        010F:   player $PLAYER_CHAR wanted_level > 0
                    then
                        0AF0: 2@ = read_int_from_ini_file "cleo\Busy Pedestrians.ini" section "Security Van Heist" key "Money Default Min"
                        0AF0: 3@ = read_int_from_ini_file "cleo\Busy Pedestrians.ini" section "Security Van Heist" key "Money Default Max"
                        0209: 14@ = random_int_in_ranges 2@ 3@
                    else
                        0AF0: 2@ = read_int_from_ini_file "cleo\Busy Pedestrians.ini" section "Security Van Heist" key "Money Stealth Min"
                        0AF0: 3@ = read_int_from_ini_file "cleo\Busy Pedestrians.ini" section "Security Van Heist" key "Money Stealth Max"
                        0209: 14@ = random_int_in_ranges 2@ 3@
                    end
                end
            02E1: Case_money = create_cash_pickup 14@ at 4@ 5@ 6@ permanence_flag 0
            0E38: get_pickup_pointer Case_money store_to 1@
            1@ += 4 // Указатель на CObject
            0A8D: 1@ = read_memory 1@ size 4 virtual_protect 0
            0AEC: 2@ = object_struct 1@ handle
            08D2: object 2@ scale_model 0.0
            model.Load(1550)
            038B: load_requested_models
            0F03: create_render_object_to_object 2@ model 1550 offset 0 0 0 rotation 0 0 0 store_to 1@
            model.Destroy(1550)

            

            0A97: 4@ = car car_security struct
            0E28: write_struct 4@ offset Memory_adr size 1 value 1 // Ставим метку для невозможности повторного кейса

            if
                check_damage == true
            then
                // Если в машине сидят то будут атаковать игрока, а не поедут дальше
                if and
                    803B:   not Security_second == $PLAYER_ACTOR  // (int)
                    056D:   actor Security_second defined
                    Security_second > 0
                then
                    if
                        00DB:   actor Security_second in_car car_security
                    then
                        05E2: AS_actor Security_second kill_actor $PLAYER_ACTOR
                        gosub @Wanted_level
                    end
                end
                if and
                    803B:   not Security_Driver == $PLAYER_ACTOR  // (int)
                    056D:   actor Security_Driver defined
                then
                    if
                        00DB:   actor Security_Driver in_car car_security
                    then
                        05E2: AS_actor Security_Driver kill_actor $PLAYER_ACTOR
                        gosub @Wanted_level
                    end
                end
            end
            break
        end
    else
        break
    end
until not player.Defined(0)

:Voice

:Remove_Voice

    timera > 5000
    not Voice == -1
then
    0AAE: remove_audio_stream Voice
    Voice = -1
    timera = 0
end

    1@ >= 22
    1@ <= 5
then
    2@ -= 1 // Уровень розыска
    5@ += 1 // Выполнено условие
end

// Головной убор
08F7: get_player $PLAYER_CHAR bodypart 16 textureCRC_to 1@ modelCRC_to 3@ // Головной убор
08F7: get_player $PLAYER_CHAR bodypart 15 textureCRC_to 3@ modelCRC_to 4@ // Очки/Маска

    1@ <> 0
    3@ <> 0
then
    2@ -= 1 // Уровень розыска
    5@ += 1 // Выполнено условие
end

if
    5@ >= 2
then
    if
        02D8:   actor $PLAYER_ACTOR current_weapon == 23 // Которое держит в руках
    then
        2@ = 0
    else
        0D59: get_current_weather_to 1@
        if or
            1@ == 8
            1@ == 16
        then
            2@ = 0
        end
    end
end
