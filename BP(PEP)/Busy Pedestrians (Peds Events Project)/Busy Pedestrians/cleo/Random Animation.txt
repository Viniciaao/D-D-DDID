        089F: get_actor 0@ pedtype_to 1@

            1@ == 4

            if
                0E4D: random_percent 70
            then
                0209: 1@ = random_int_in_ranges 1 5
                0AD3: 5@v = string_format "WalkIdle%d" 1@

                    02F2:   actor 0@ model == 31 //
                    02F2:   actor 0@ model == 53 //
                    02F2:   actor 0@ model == 88 //
                    02F2:   actor 0@ model == 160 //
                then
                    0605: actor 0@ perform_animation "old_man_1" IFP "PedEventsMod" framedelta 2.0 loop 0 lockX 1 lockY 1 lockF 0 time -1
                else
                    0605: actor 0@ perform_animation "TripUp" IFP "PedEventsMod" framedelta 2.0 loop 0 lockX 1 lockY 1 lockF 0 time -1
                end
            end

            wait 1000
            if
                056D:   actor 0@ defined

                0EFF: get_char_simplest_active_task 0@ id_to 1@ pointer_to 2@
                if
                    1@ == 203

                    05DE: AS_actor 0@ walk_around_ped_path
