        if or
            actor.Dead(bum_man)
            2@ == 11
            3@ == 49

            051A:   actor bum_man damaged_by_actor $PLAYER_ACTOR
            not is_char_script_controlled bum_man
            not $ACTIVE_INTERIOR == 0
        then
            break
        end
        
        080E: get_actor bum_man ped_event_to 2@
        if or
            2@ == 15
            2@ == 31
            2@ == 59
        then
            break
        end
        
        // Погода и время
        0E40: get_current_hour 2@
        if or
            not 2@ >= 9
            not 20 >= 2@
        then
            break
        end
        
        // Если целится из огнестрела

            0457:   player $PLAYER_CHAR aiming_at_actor bum_man

                        8A2A:   not text_box 'donat2' displayed
                    then

                not Event > 0

                00E1:   player 0 pressed_key 15
