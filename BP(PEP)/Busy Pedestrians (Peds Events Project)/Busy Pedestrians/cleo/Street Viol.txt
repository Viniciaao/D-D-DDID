                        then 0AAE: remove_audio_stream Music_stream
                        end
                        Music_stream = -1
                    end
                end

                // Запуск или смена песни
                if
                    Progress == 1
                then
                    if not Music_stream == -1
                    then 0AAE: remove_audio_stream Music_stream
                    end
                    Music_stream = -1
                    0209: 2@ = random_int_in_ranges 1 Max_Song
                    if 003B:   2@ == Not_Again_this_song  // (int)
                    then continue
                    end
                    0085: Not_Again_this_song = 2@ // (int)
                    0AD3: 3@v = string_format "M\V%d.mp3" 2@
                    0AC1: Music_stream = load_audio_stream_with_3d_support 3@v // IF and SET
                    0AC4: link_3d_audio_stream Music_stream to_actor Musician
                    0AAD: set_audio_stream Music_stream state 1
                    progress = 2
                    // Если анимация была прервана
                    if
                        8611:   not actor Musician performing_animation "viol1"
                    then
                        0812: AS_actor Musician perform_animation "viol1" IFP "PedEventsMod" framedelta 2.0 loopA 1 lockX 0 lockY 0 lockF 1 time -1
                    end
                end

                // Все основное действие
                if
                    progress == 2
                then

                    // Если песня закончилась
                    0AB9: get_audio_stream Music_stream state_to 2@
                    if 2@ == -1
                    then
                        progress = 1
                        continue
                    end

                    // Громкость от камеры
                    04C4: store_coords_to 2@ 3@ 4@ from_actor Musician with_offset 0.0 0.0 0.0
                    0EB2: get_offset_from_camera_in_world_coords 0.0 0.0 0.0 store_to 5@ 6@ 7@
                    050A: 3@ = distance_between_XYZ 2@ 3@ 4@ and_XYZ 5@ 6@ 7@
                    3@ /= 100.0
                    0087: 2@ = Volume_song // Громкость из .ini файла
                    0063: 2@ -= 3@  // (float)
                    if 2@ < 0.0
                    then 2@ = 0.0
                    end
                    0ABC: set_audio_stream Music_stream volume 2@

                    // Если не выполняет анимацию (например вмешался другйо скрипт) то останавливаем музыку
                    0AB9: get_audio_stream Music_stream state_to 2@
                    if and
                        event == 0
                        Event_ped == 0
                        8611:   not actor Musician performing_animation "viol1"
                        8611:   not actor Musician performing_animation "viol2"
                    then
                        0AAD: set_audio_stream Music_stream state 2
                    else
                        if 2@ == 2
                        then 0AAD: set_audio_stream Music_stream state 1
                        end
                    end

                end
            else
                break
            end
        until not player.Defined(0)
        // Выход из цикла

        if not Music_stream == -1
        then 0AAE: remove_audio_stream Music_stream
        end

        // Если далеко то удаляем все, если нет то освобождаем от скрипта
        if
            08B7:   test Bit_Check bit Destroy_all
        then
            if 056D:   actor Musician defined
            then 009B: destroy_actor Musician
            end
        else
            if 056D:   actor Musician defined
            then 01C2: remove_references_to_actor Musician // Like turning an actor into a random pedestrian
            end
        end


        04EF: release_animation "PedEventsMod"

        repeat
        wait 0
        until not $ACTIVE_INTERIOR == 18
    
    end


        end
    end
end
